# IPv4 extraction tests
#
# Each case is two lines:
#   input: <the exact line fed to the program>
#   output: <the result line, not the prompt>
# A blank input is written as "input:" with nothing after it.
# "END" is last. It quits instead of extracting an address.
# Lines starting with # are comments.
#
# Decimal values were computed as (A<<24)|(B<<16)|(C<<8)|D, independently of
# the C++ program.

# --- sample run ---

input: connecting to 192.168.1.1 now
output: Extracted IPv4 address: 192.168.1.1 (decimal value: 3232235777, port: none)

input: server=10.0.0.255:8080end
output: Extracted IPv4 address: 10.0.0.255 (decimal value: 167772415, port: 8080)

input: 192a168.1.1.1
output: Extracted IPv4 address: 168.1.1.1 (decimal value: 2818638081, port: none)

input: 192.168.1.1.
output: Invalid input: no valid IPv4 address found

input: Connection from 192.168.1.1 refused
output: Extracted IPv4 address: 192.168.1.1 (decimal value: 3232235777, port: none)

input: 192.168.01.1
output: Invalid input: no valid IPv4 address found

input: 1.2.3.4:99999
output: Invalid input: no valid IPv4 address found

input: 12.34.56
output: Invalid input: no valid IPv4 address found

input: no number here
output: Invalid input: no valid IPv4 address found

# --- leading zeros: only a single 0 is legal ---

input: 0.0.0.0
output: Extracted IPv4 address: 0.0.0.0 (decimal value: 0, port: none)

input: 0.0.0.0:0
output: Extracted IPv4 address: 0.0.0.0 (decimal value: 0, port: 0)

input: 00.0.0.0
output: Invalid input: no valid IPv4 address found

input: 01.2.3.4
output: Invalid input: no valid IPv4 address found

input: 1.2.3.04
output: Invalid input: no valid IPv4 address found

input: 1.2.3.4:0
output: Extracted IPv4 address: 1.2.3.4 (decimal value: 16909060, port: 0)

input: 1.2.3.4:00
output: Invalid input: no valid IPv4 address found

input: 1.2.3.4:01
output: Invalid input: no valid IPv4 address found

input: 0.0.0.0:00
output: Invalid input: no valid IPv4 address found

input: 1.2.3.4:00000
output: Invalid input: no valid IPv4 address found

input: 10.20.30.40:080
output: Invalid input: no valid IPv4 address found

# --- range limits ---

input: 255.255.255.255:65535
output: Extracted IPv4 address: 255.255.255.255 (decimal value: 4294967295, port: 65535)

input: 1.2.3.4:65535
output: Extracted IPv4 address: 1.2.3.4 (decimal value: 16909060, port: 65535)

input: 1.2.3.4:65536
output: Invalid input: no valid IPv4 address found

input: 1.2.3.4:123456
output: Invalid input: no valid IPv4 address found

input: 1.2.3.4:9999
output: Extracted IPv4 address: 1.2.3.4 (decimal value: 16909060, port: 9999)

input: 256.1.1.1
output: Invalid input: no valid IPv4 address found

input: 255.255.255.256
output: Invalid input: no valid IPv4 address found

input: 999.1.1.1
output: Invalid input: no valid IPv4 address found

input: 1234.5.6.7
output: Invalid input: no valid IPv4 address found

# --- the whole run must match; do not keep a prefix ---

input: 192.168.1.1.1
output: Invalid input: no valid IPv4 address found

input: 1.2.3.4:
output: Invalid input: no valid IPv4 address found

input: 1.2.3.4:80:80
output: Invalid input: no valid IPv4 address found

input: 1.2.3.4::80
output: Invalid input: no valid IPv4 address found

input: 192.168.1.1:80.1
output: Invalid input: no valid IPv4 address found

input: :1.2.3.4
output: Invalid input: no valid IPv4 address found

input: .1.2.3.4
output: Invalid input: no valid IPv4 address found

input: 1..2.3.4
output: Invalid input: no valid IPv4 address found

input: 192.168.1
output: Invalid input: no valid IPv4 address found

input: 1.2.3.4.
output: Invalid input: no valid IPv4 address found

# A colon glued on with no port rejects the address. A space is a separator,
# so the address before the space is its own run and is accepted with no port.

input: 192.168.1.1:
output: Invalid input: no valid IPv4 address found

input: 192.168.1.1 :80
output: Extracted IPv4 address: 192.168.1.1 (decimal value: 3232235777, port: none)

# --- separators, and which run wins ---

input: +1.2.3.4
output: Extracted IPv4 address: 1.2.3.4 (decimal value: 16909060, port: none)

input: 192.168.1.1/24
output: Extracted IPv4 address: 192.168.1.1 (decimal value: 3232235777, port: none)

input: v1.2.3.4
output: Extracted IPv4 address: 1.2.3.4 (decimal value: 16909060, port: none)

input: 1.2.3.4 and 5.6.7.8
output: Extracted IPv4 address: 1.2.3.4 (decimal value: 16909060, port: none)

input: 192.168.1.1a10.0.0.1
output: Extracted IPv4 address: 192.168.1.1 (decimal value: 3232235777, port: none)

# The first run is rejected, so the scan continues and takes the later address.

input: 1.2.3.4:99999 8.8.8.8
output: Extracted IPv4 address: 8.8.8.8 (decimal value: 134744072, port: none)

input: 1.2.3.4:65535 extra
output: Extracted IPv4 address: 1.2.3.4 (decimal value: 16909060, port: 65535)

input: 1.2.3.4:65535extra
output: Extracted IPv4 address: 1.2.3.4 (decimal value: 16909060, port: 65535)

input: 10.20.30.40:80
output: Extracted IPv4 address: 10.20.30.40 (decimal value: 169090600, port: 80)

input: host.example.com
output: Invalid input: no valid IPv4 address found

input:
output: Invalid input: no valid IPv4 address found

# The input below is three spaces.
input:    
output: Invalid input: no valid IPv4 address found

# "end" is not the quit command. Only the exact line END is.

input: end
output: Invalid input: no valid IPv4 address found

input: END
output: Program terminated.
